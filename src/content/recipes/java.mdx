      public class Sieve {
          public static void main(String[] args) {
              int n = 100;
              boolean[] composite = new boolean[n + 1];
              for (int p = 2; p * p <= n; p++) {
                  if (!composite[p]) {
                      for (int m = p * p; m <= n; m += p) composite[m] = true;
                  }
              }
              int count = 0;
              for (int i = 2; i <= n; i++) {
                  if (!composite[i]) {
                      System.out.print(i + " ");
                      count++;
                  }
              }
              System.out.println("\n" + count + " primes below " + n);
          }
      }

    note: "Halving the interval — and the midpoint that avoids overflow."
    group: "Algorithms"

      public class BinarySearch {
          static int find(int[] a, int target) {
              int lo = 0, hi = a.length - 1;
              while (lo <= hi) {
                  int mid = lo + (hi - lo) / 2;   // not (lo + hi) / 2, which can overflow
                  if (a[mid] == target) return mid;
                  if (a[mid] < target) lo = mid + 1;
                  else                 hi = mid - 1;
              }
              return -1;
          }

          public static void main(String[] args) {
              int[] a = {2, 5, 8, 12, 16, 23, 38, 56, 72, 91};
              for (int t : new int[]{23, 2, 91, 7}) {
                  System.out.println(t + " -> index " + find(a, t));
              }
          }
      }
