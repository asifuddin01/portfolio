      #include <string.h>

      void reverse(char *s) {
          size_t i = 0, j = strlen(s);
          if (j == 0) return;
          for (j--; i < j; i++, j--) {
              char t = s[i];
              s[i] = s[j];
              s[j] = t;
          }
      }

      int main(void) {
          char s[] = "computational graph";
          printf("before %s\n", s);
          reverse(s);
          printf("after  %s\n", s);
          return 0;
      }

      typedef struct Node { int value; struct Node *next; } Node;

      Node *push(Node *head, int v) {
          Node *n = malloc(sizeof *n);
          n->value = v;
          n->next = head;
          return n;
      }

      void show(Node *head) {
          for (Node *p = head; p; p = p->next) printf("%d -> ", p->value);
          printf("NULL\n");
      }

      void destroy(Node *head) {
          while (head) { Node *next = head->next; free(head); head = next; }
      }

      int main(void) {
          Node *list = NULL;
          for (int i = 1; i <= 5; i++) list = push(list, i * 10);
          show(list);
          destroy(list);
          return 0;
      }

    note: "Two classics written out in full."

    code: |2-
      #include <stdio.h>

      void bubble(int *xs, int n) {
          for (int i = 0; i < n - 1; i++)
              for (int j = 0; j < n - 1 - i; j++)
                  if (xs[j] > xs[j+1]) { int t = xs[j]; xs[j] = xs[j+1]; xs[j+1] = t; }
      }

      int search(const int *xs, int n, int target) {
          int lo = 0, hi = n - 1;
          while (lo <= hi) {
              int mid = lo + (hi - lo) / 2;      /* avoids overflow */
              if (xs[mid] == target) return mid;
              if (xs[mid] <  target) lo = mid + 1;
              else                   hi = mid - 1;
          }
          return -1;
      }

      int main(void) {
          int xs[] = {9, 3, 7, 1, 8, 2};
          int n = sizeof xs / sizeof xs[0];

          bubble(xs, n);
          for (int i = 0; i < n; i++) printf("%d ", xs[i]);
          printf("\n");

          for (int t = 1; t <= 4; t++) printf("search %d -> %d\n", t, search(xs, n, t));
          return 0;
      }

    group: "Algorithms"

      #include <stdio.h>
      #include <stdlib.h>
      #include <string.h>

      int main(void) {
          int limit = 100;
          char *composite = calloc(limit + 1, 1);

          for (int p = 2; (long)p * p <= limit; p++)
              if (!composite[p])
                  for (int m = p * p; m <= limit; m += p) composite[m] = 1;

          int count = 0;
          for (int n = 2; n <= limit; n++)
              if (!composite[n]) { printf("%3d", n); if (++count % 10 == 0) printf("\n"); }

          printf("\n%d primes below %d\n", count, limit);
          free(composite);
          return 0;
      }
